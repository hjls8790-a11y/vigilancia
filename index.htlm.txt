<body>

    <div id="status-indicator">Estado: En espera</div>
    <div id="map"></div>

    <div id="report-form">
        <h3>Nuevo Reporte</h3>
        <input type="text" id="comment" placeholder="Escribe un comentario...">
        <input type="file" id="photo" accept="image/*" capture="camera">
        <button class="btn btn-start" onclick="saveReport()">Guardar Reporte</button>
        <button class="btn" style="background:#ccc; margin-top:5px; color:#333" onclick="closeForm()">Cancelar</button>
    </div>

    <div id="control-panel">
        <button id="btnStart" class="btn btn-start" onclick="startPatrol()">Iniciar Rondín</button>
        <button id="btnReport" class="btn btn-report" onclick="openForm()">📷 Agregar Novedad / Foto</button>
        <button id="btnStop" class="btn btn-stop" onclick="stopPatrol()">Finalizar Rondín</button>
    </div>

    <script>
        let map;
        let polyline;
        let path = [];
        let watchId = null;
        let isPatrolling = false;
        let currentReportPos = null;

        // Inicializar Mapa Libre centrado en Culiacán (cambiará al detectar GPS)
        map = L.map('map').setView([24.8016, -107.3934], 15);
        
        L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
            attribution: '© OpenStreetMap'
        }).addTo(map);

        polyline = L.polyline(path, {color: 'blue', weight: 5}).addTo(map);

        function startPatrol() {
            if (!navigator.geolocation) {
                alert("ERROR: Tu navegador no soporta geolocalización.");
                return;
            }

            document.getElementById("status-indicator").innerText = "Estado: BUSCANDO SEÑAL GPS...";
            document.getElementById("status-indicator").style.backgroundColor = "orange";
            
            document.getElementById("btnStart").style.display = "none";
            document.getElementById("btnStop").style.display = "block";
            document.getElementById("btnReport").style.display = "block";

            watchId = navigator.geolocation.watchPosition(
                (position) => {
                    isPatrolling = true;
                    document.getElementById("status-indicator").innerText = "Estado: VIGILANCIA ACTIVA";
                    document.getElementById("status-indicator").style.backgroundColor = "red";
                    
                    const lat = position.coords.latitude;
                    const lng = position.coords.longitude;
                    
                    path.push([lat, lng]);
                    polyline.setLatLngs(path);
                    map.setView([lat, lng], 18); // Centrar y acercar el mapa
                },
                (error) => {
                    if(error.code == 1) alert("ERROR: Debes dar permiso de ubicación.");
                    if(error.code == 2) alert("ERROR: No hay señal de GPS.");
                    
                    document.getElementById("status-indicator").innerText = "Estado: ERROR GPS";
                    document.getElementById("status-indicator").style.backgroundColor = "black";
                    
                    document.getElementById("btnStart").style.display = "block";
                    document.getElementById("btnStop").style.display = "none";
                    document.getElementById("btnReport").style.display = "none";
                },
                { enableHighAccuracy: true, timeout: 10000, maximumAge: 0 }
            );
        }

        function stopPatrol() {
            isPatrolling = false;
            if (watchId !== null) navigator.geolocation.clearWatch(watchId);
            
            document.getElementById("status-indicator").innerText = "Estado: Rondín Finalizado";
            document.getElementById("status-indicator").style.backgroundColor = "green";

            document.getElementById("btnStart").style.display = "block";
            document.getElementById("btnStop").style.display = "none";
            document.getElementById("btnReport").style.display = "none";

            alert("Rondín finalizado. Puntos registrados: " + path.length);
        }

        function openForm() {
            navigator.geolocation.getCurrentPosition(
                (position) => {
                    currentReportPos = [position.coords.latitude, position.coords.longitude];
                    document.getElementById("report-form").style.display = "block";
                },
                (error) => { alert("No se pudo obtener la ubicación para el reporte."); },
                { enableHighAccuracy: true }
            );
        }

        function closeForm() {
            document.getElementById("report-form").style.display = "none";
            document.getElementById("comment").value = "";
            document.getElementById("photo").value = "";
        }

        function saveReport() {
            const comment = document.getElementById("comment").value;
            const fileInput = document.getElementById("photo");
            const file = fileInput.files[0];

            if (!comment && !file) {
                alert("Escribe un comentario o toma una foto.");
                return;
            }

            const reader = new FileReader();
            
            const createMarker = (imgSrc) => {
                const content = <b>Reporte:</b> ${comment}<br>${imgSrc ? `<img src="${imgSrc}" style="width:100px; margin-top:5px;"> : ''}`;
                L.marker(currentReportPos).addTo(map).bindPopup(content);
                closeForm();
            };

            if (file) {
                reader.onloadend = function() { createMarker(reader.result); }
                reader.readAsDataURL(file);
            } else {
                createMarker(null);
            }
        }
    </script>
